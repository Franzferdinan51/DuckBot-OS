def route(task, prompt, max_hops=3):
    attempts = 0
    conf_guess = score(task)
    out = None
    for tier in TIERS:
        if attempts >= max_hops and task.get("risk","low") == "low": break

_cache = {}  # key -> (expires_ts, result)
_breaker = {}  # tier_name -> reopen_ts

def _cache_key(task, prompt):
    sig = hashlib.sha256((task.get("kind","*")+"|"+task.get("risk","low")+"|"+prompt).encode()).hexdigest()
    return sig

def _get_cache(key):
    hit = _cache.get(key)
    if not hit: return None
    exp, val = hit
    if exp > time.time(): return val
    _cache.pop(key, None)
    return None

def _set_cache(key, val, ttl=TTL_CACHE):
    _cache[key] = (time.time()+ttl, val)

    return _breaker.get(tier, 0) > time.time()

# ---------- main entry ----------
def route_task(task):
    """
    task = {
      "kind": "summary|status|json_format|code|policy|long_form|arbiter|reasoning|*",
      "risk": "low|medium|high",
      "prompt": "string...",
    }
    """
    prompt = task.get("prompt","").strip()
    key = _cache_key(task, prompt)
    cached = _get_cache(key)
    if cached: return {**cached, "cached": True}

    hops = 0
    conf_guess = _initial_confidence(task)
    chosen = None
    out = None

    for tier_name in _ladder(task):
        if hops >= _max_hops(task) and task.get("risk","low") == "low":
            break
        if _blocked(tier_name): 
            continue

            out = _call_tier(tier_name, prompt)
            conf = float(out.get("confidence", conf_guess))
            out["model_used"] = tier_name
            chosen = tier_name

            if conf >= CONF_MIN:
                _set_cache(key, out)
                return out

                out = _call_tier(tier_name, prompt)
                out["model_used"] = tier_name
                _set_cache(key, out)
                return out
            except TimeoutError:
                _trip(tier_name)
